import numpy as np

class LogisticRegression:
    def __init__(self, learning_rate=0.1, iterations=1000):
        self.lr = learning_rate
        self.iterations = iterations
        self.weights = None
        self.bias = None

    def _sigmoid(self, z):
        return 1 / (1 + np.exp(-z))

    def fit(self, X, y):
        n_samples, n_features = X.shape
        self.weights = np.zeros(n_features)
        self.bias = 0

        for _ in range(self.iterations):
            linear_model = np.dot(X, self.weights) + self.bias
            y_predicted = self._sigmoid(linear_model)

            # Calculate gradients
            dw = (1 / n_samples) * np.dot(X.T, (y_predicted - y))
            db = (1 / n_samples) * np.sum(y_predicted - y)

            # Update parameters
            self.weights -= self.lr * dw
            self.bias -= self.lr * db

    def predict(self, X):
        linear_model = np.dot(X, self.weights) + self.bias
        y_predicted = self._sigmoid(linear_model)
        return np.array([1 if i > 0.5 else 0 for i in y_predicted])

    def confusion_matrix(self, y_true, y_pred):
        tp = tn = fp = fn = 0
        for actual, predicted in zip(y_true, y_pred):
            if actual == 1 and predicted == 1:
                tp += 1
            elif actual == 0 and predicted == 0:
                tn += 1
            elif actual == 0 and predicted == 1:
                fp += 1
            elif actual == 1 and predicted == 0:
                fn += 1
        
        return np.array([[tn, fp], [fn, tp]])

# --- Data Initialization ---
X = np.array([[5.50], [3.25], [4.25], [2.00], [1.50], [0.50], [7.00]])
y = np.array([1, 1, 1, 0, 0, 0, 1])

# --- Model Execution ---
model = LogisticRegression(learning_rate=0.1, iterations=1000)
model.fit(X, y)
predictions = model.predict(X)
matrix = model.confusion_matrix(y, predictions)

print(f"Predictions: {predictions}")
print("Confusion Matrix:")
print(matrix)

#RESIZING IMAGES
import cv2
import os
import numpy as np

# Configuration
BASE_PATH = r"C:\Users\simple_man\Downloads\CATS AND DOGS"
CATEGORIES = ["Cat", "Dog"]
IMG_SIZE = 50
data = []

def create_training_data():
    for category in CATEGORIES:
        path = os.path.join(BASE_PATH, category)
        class_num = CATEGORIES.index(category)  # 0 for Cat, 1 for Dog
        
        for img in os.listdir(path):
            try:
                # Load image in grayscale to save memory (common for 50x50)
                # or remove cv2.IMREAD_GRAYSCALE if you want color
                img_array = cv2.imread(os.path.join(path, img), cv2.IMREAD_GRAYSCALE)
                
                # Resize to 50 x 50
                new_array = cv2.resize(img_array, (IMG_SIZE, IMG_SIZE))
                
                # Store image and its label
                data.append([new_array, class_num])
            except Exception as e:
                # Skip broken or corrupt images
                pass

create_training_data()
print(f"Successfully processed {len(data)} images to size {IMG_SIZE}x{IMG_SIZE}.")

General Comments

  • On model performance the SVM is a traditional classifier that works by finding a boundary in flattened pixel data. While it is fast for small dataset of 50 images. it lacks the spatial intelligence of deep learning mode; to detect textures like features like ears
  • Resizing to 50 X 50 and using only 50 samples limits the model's accuracy. At this scale , the model is likely memorizing spefic pixel patterns ( overfitting) rather than learning the actual difference between a cat and a dog.
